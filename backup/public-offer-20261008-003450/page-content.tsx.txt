function contactHref(offer = false) {
  const text = offer ? `Hello, I would like to discuss the demo and offer for ${content.offer?.title}. Please confirm eligibility and pricing.` : "Hello, I would like to discuss a website or online promotion for my business. Please help me understand the suitable services and costs.";
  return content.whatsapp ? `https://wa.me/${content.whatsapp.replace(/\D/g, "")}?text=${encodeURIComponent(text)}` : `mailto:${email}?subject=${encodeURIComponent(offer ? "Starter offer & demo enquiry" : "Business website & promotion enquiry")}&body=${encodeURIComponent(text)}`;
}
function ContactButtons({ offer = false }: { offer?: boolean }) {

function Countdown({ end }: { end: string }) {
  const [remaining, setRemaining] = useState<number | null>(null);
  useEffect(() => { const tick = () => setRemaining(Math.max(0, Date.parse(end) - Date.now())); tick(); const timer = setInterval(tick, 1000); return () => clearInterval(timer); }, [end]);
  if (!Number.isFinite(Date.parse(end))) return <p>Contact us for current offer details.</p>;
  if (remaining === null) return <p>Offer deadline: {new Date(end).toLocaleString("en-IN", { timeZone: "Asia/Kolkata" })} IST</p>;
  if (!remaining) return <p>This offer has ended. Contact us for current pricing.</p>;
  const total = Math.floor(remaining / 1000);
  return <div className={s.timer} aria-label="Offer time remaining">{[[Math.floor(total / 86400), "Days"], [Math.floor(total / 3600) % 24, "Hours"], [Math.floor(total / 60) % 60, "Minutes"], [total % 60, "Seconds"]].map(([value, label]) => <div key={label}><b>{String(value).padStart(2, "0")}</b><span>{label}</span></div>)}</div>;
}

function StarterOffer() {
  const [now, setNow] = useState<number | null>(null);
  useEffect(() => { if (!content.offer) return; const tick = () => setNow(Date.now()); tick(); const timer = setInterval(tick, 1000); return () => clearInterval(timer); }, []);
  const offer = content.offer;
  if (!offer) return null;
  const active = now !== null && Number.isFinite(Date.parse(offer.endsAt)) && now < Date.parse(offer.endsAt);
  return <section className={s.offer}><span className={s.label}>STARTER OFFER</span><h2>{offer.title}</h2>{active ? <><p><del>{money(offer.regularPrice)}</del> · Offer {money(offer.offerPrice)}</p><strong>{money(offer.contactPrice)} <small>when you contact us before the deadline</small></strong><Countdown end={offer.endsAt}/><ul>{offer.inclusions.map(item => <li key={item}>{item}</li>)}</ul><p>{offer.demoDescription}</p><ContactButtons offer/><small>{offer.terms}</small></> : <><Countdown end={offer.endsAt}/><ContactButtons/></>}</section>;
}
