// Disposable browser QA: no production content or real portal accounts are changed.
import assert from 'node:assert/strict';
import http from 'node:http';
import net from 'node:net';
import path from 'node:path';
import { fileURLToPath } from 'node:url';
import { mkdirSync, readFileSync, writeFileSync, mkdtempSync, symlinkSync, copyFileSync, cpSync, rmSync } from 'node:fs';
import { randomBytes } from 'node:crypto';
import { spawn } from 'node:child_process';
const root = path.resolve(path.dirname(fileURLToPath(import.meta.url)), '..');

if (process.argv.includes('--serve')) {
  const { default: next } = await import('next');
  const app = next({ dev: true, webpack: true, dir: process.env.CONTENT_QA_ROOT, conf: { distDir: '.next' } });
  await app.prepare();
  const handler = app.getRequestHandler();
  const server = http.createServer((req, res) => handler(req, res));
  await new Promise(resolve => server.listen(Number(process.env.PORT), '127.0.0.1', resolve));
  process.send?.('ready');
  process.on('message', async message => { if (message === 'stop') { server.close(); await app.close(); process.exit(0); } });
} else {
  const { MongoMemoryServer } = await import('mongodb-memory-server');
  const { default: mongoose } = await import('mongoose');
  const { chromium } = await import('playwright');
  const { SignJWT } = await import('jose');
  let database, portal, browser, app, checkout;
  let output = '';
  try {
    database = await MongoMemoryServer.create({ instance: { launchTimeout: 60000 } });
    await mongoose.connect(database.getUri(), { dbName: 'page_content_browser' });
    browser = await chromium.launch();
    const fixturePage = await browser.newPage({ viewport: { width: 900, height: 1200 } });
    await fixturePage.setContent('<body style="margin:0;background:#00243d;color:#66e4ee;font-family:Arial;padding:50px"><h1>Content QA image</h1><p>Portrait media fixture</p></body>');
    const fixture = await fixturePage.screenshot(); await fixturePage.close();
    const userId = '1'.repeat(24);
    portal = http.createServer((req, res) => {
      if (req.url === '/fixture.png') { res.writeHead(200, { 'Content-Type': 'image/png' }); res.end(fixture); return; }
      res.setHeader('Content-Type', 'application/json');
      if (req.url === '/api/user-details') res.end(JSON.stringify({ success: true, data: { _id: userId, name: 'Content Test Editor', email: 'editor@content.test', roles: ['admin'], isGuest: false } }));
      else { res.statusCode = 404; res.end('{}'); }
    });
    await new Promise(resolve => portal.listen(0, '127.0.0.1', resolve));
    const portalOrigin = `http://127.0.0.1:${portal.address().port}`;
    const assetId = new mongoose.Types.ObjectId();
    await mongoose.connection.db.collection('mediaassets').insertOne({ _id: assetId, url: `${portalOrigin}/fixture.png`, publicId: 'content-test-fixture', assetId: 'fixture', assetFolder: 'test', width: 900, height: 1200, bytes: fixture.length, format: 'png', altText: 'Content QA image', kind: 'image', createdAt: new Date(), updatedAt: new Date() });
    const portServer = net.createServer(); await new Promise(resolve => portServer.listen(0, '127.0.0.1', resolve)); const port = portServer.address().port; await new Promise(resolve => portServer.close(resolve));
    const origin = `http://127.0.0.1:${port}`;
    const secret = randomBytes(32).toString('hex');
    mkdirSync(path.join(root, '.next'), { recursive: true });
    checkout = mkdtempSync(path.join(root, '.next', 'content-browser-'));
    cpSync(path.join(root, 'src'), path.join(checkout, 'src'), { recursive: true });
    // The disposable preview avoids remote font fetching; actual page/editor code and CSS are unchanged.
    writeFileSync(path.join(checkout, 'src/app/layout.tsx'), 'import "./globals.css"; export default function Layout({children}:{children:React.ReactNode}) { return <html lang="en"><body>{children}</body></html>; }');
    for (const name of ['public', 'node_modules']) symlinkSync(path.join(root, name), path.join(checkout, name), process.platform === 'win32' ? 'junction' : 'dir');
    for (const name of ['package.json', 'tsconfig.json']) copyFileSync(path.join(root, name), path.join(checkout, name));
    app = spawn(process.execPath, [fileURLToPath(import.meta.url), '--serve'], { cwd: checkout, stdio: ['ignore', 'pipe', 'pipe', 'ipc'], windowsHide: true, env: { ...process.env, CONTENT_QA_ROOT: checkout, PORT: String(port), NODE_ENV: 'development', DATA_DRIVER: 'mongo', MONGODB_URI: database.getUri(), MONGODB_DB: 'page_content_browser', PORTAL_API_URL: portalOrigin, PORTAL_URL: portalOrigin, TOKEN_SECRET_KEY: secret, VERCEL_ENV: 'preview', INDEXNOW_KEY: '', NEXT_TELEMETRY_DISABLED: '1' } });
    app.stdout.on('data', chunk => { output += chunk; }); app.stderr.on('data', chunk => { output += chunk; });
    await new Promise((resolve, reject) => { const timer = setTimeout(() => reject(Error('QA server startup timed out')), 120000); app.once('message', () => { clearTimeout(timer); resolve(); }); app.once('exit', code => { clearTimeout(timer); reject(Error(`QA server exited ${code}`)); }); });
    console.log('Disposable QA server ready');
    const token = await new SignJWT({ _id: userId }).setProtectedHeader({ alg: 'HS256' }).setExpirationTime('1h').sign(new TextEncoder().encode(secret));
    const editor = await browser.newContext({ viewport: { width: 1440, height: 1000 } });
    await editor.addCookies([{ name: 'token', value: token, url: origin }]);
    const page = await editor.newPage(); page.setDefaultTimeout(60000);
    const errors = []; page.on('pageerror', error => errors.push(error.message));
    const defaults = JSON.parse(readFileSync(path.join(root, 'src/lib/page-content-defaults.json'), 'utf8'));
    const location = (site, section) => `${origin}/admin/content?page=${site}&section=${section}`;
    const save = async () => {
      const request = page.waitForResponse(response => response.url().endsWith('/api/website-content') && response.request().method() === 'PATCH');
      await page.getByRole('button', { name: 'Save changes', exact: true }).click();
      const response = await request; assert.equal(response.status(), 200, await response.text());
      await page.getByText('Section saved. Changes are live on the website.', { exact: true }).waitFor();
    };
    const initialResponse = await page.goto(location('about', 'hero'), { waitUntil: 'domcontentloaded', timeout: 300000 });
    assert.equal(initialResponse.status(), 200, 'Editor must render successfully');
    assert.equal(await page.getByLabel('Main heading', { exact: true }).inputValue(), defaults.about.hero.heading);
    await page.getByLabel('Main heading', { exact: true }).fill('QA saved heading');
    await page.getByRole('button', { name: 'Choose image, GIF or MP4', exact: true }).click();
    await page.getByRole('button', { name: /Content QA image/ }).click();
    await page.getByLabel('Frame size', { exact: true }).selectOption('source');
    await save(); await page.reload();
    assert.equal(await page.getByLabel('Main heading', { exact: true }).inputValue(), 'QA saved heading');
    assert.equal(await page.getByLabel('Frame size', { exact: true }).inputValue(), 'source');
    const publicPage = await editor.newPage(); await publicPage.goto(`${origin}/about`, { waitUntil: 'domcontentloaded' });
    assert.match(await publicPage.locator('h1').innerText(), /QA saved heading/);
    const ratio = await publicPage.locator('main .section-visual').first().evaluate(node => getComputedStyle(node).aspectRatio);
    assert.equal(ratio, '900 / 1200');
    console.log('About content, image and source proportions persist after reload');
    await page.goto(location('about', 'leadership'));
    await page.getByRole('button', { name: 'Move Team members 1 down', exact: true }).click();
    await save();
    const team = (await mongoose.connection.db.collection('settings').findOne({ _id: 'site' })).pageContent.about.leadership.members;
    assert.equal(team[0].id, 'sahilpreet');
    await page.getByRole('button', { name: 'Add team member +', exact: true }).click();
    await page.getByLabel('Name', { exact: true }).last().fill('QA New Member');
    await save();
    await publicPage.reload(); assert.equal(await publicPage.getByRole('heading', { name: 'QA New Member', exact: true }).count(), 1);
    await page.goto(location('services', 'catalogue'));
    await page.getByLabel('Title', { exact: true }).nth(1).fill('QA website service');
    await save();
    await publicPage.goto(`${origin}/services`); await publicPage.getByRole('heading', { name: 'QA website service', exact: true }).waitFor();
    await publicPage.getByRole('heading', { name: 'QA website service', exact: true }).click();
    assert.equal(await publicPage.getByText(defaults.services.catalogue.groups[0].items[0].description, { exact: true }).isVisible(), true);
    const anon = await browser.newContext();
    const denied = await anon.request.patch(`${origin}/api/website-content`, { data: { page: 'about', section: 'hero', content: defaults.about.hero } }); assert.equal(denied.status(), 401);
    const hostile = await editor.request.patch(`${origin}/api/website-content`, { headers: { Origin: 'https://untrusted.example' }, data: {} }); assert.equal(hostile.status(), 403);
    const invalid = await editor.request.patch(`${origin}/api/website-content`, { data: { page: 'services', section: 'contact', content: { ...defaults.services.contact, buttonHref: 'javascript:alert(1)' } } }); assert.equal(invalid.status(), 400);
    const wrongVisual = await editor.request.patch(`${origin}/api/website-content`, { data: { page: 'about', section: 'purpose', content: defaults.about.purpose, visuals: { 'home.hero': { mode: 'pattern', pattern: 'grid', intensity: 50, fit: 'contain', focalX: 50, focalY: 50, alt: '' } } } }); assert.equal(wrongVisual.status(), 400);
    for (const [site, sections] of Object.entries(defaults)) for (const section of Object.keys(sections)) {
      await page.goto(location(site, section)); assert.equal(await page.getByText('Not yet editable', { exact: true }).count(), 0);
    }
    mkdirSync(path.join(root, 'test-results/page-content'), { recursive: true });
    for (const width of [1440, 390]) {
      await page.setViewportSize({ width, height: 1000 }); await page.goto(location('services', 'hero'));
      assert.equal(await page.evaluate(() => document.documentElement.scrollWidth > innerWidth), false, 'Admin horizontal overflow');
      await page.screenshot({ path: path.join(root, `test-results/page-content/editor-${width}.png`), fullPage: true });
      await publicPage.setViewportSize({ width, height: 1000 }); await publicPage.goto(`${origin}/services`);
      assert.equal(await publicPage.evaluate(() => document.documentElement.scrollWidth > innerWidth), false, 'Public horizontal overflow');
    }
    assert.deepEqual(errors, []);
    console.log('PASS: all sections editable; team add/reorder; service edits; permissions; validation; desktop/mobile layout');
  } catch (error) { console.error(output.slice(-12000)); throw error; }
  finally {
    if (browser) await browser.close();
    if (app?.connected) { app.send('stop'); await Promise.race([new Promise(resolve => app.once('exit', resolve)), new Promise(resolve => setTimeout(resolve, 5000))]); }
    if (app && app.exitCode === null) { if (process.platform === 'win32') { const killer = spawn('taskkill', ['/pid', String(app.pid), '/T', '/F'], { windowsHide: true, stdio: 'ignore' }); await new Promise(resolve => killer.once('exit', resolve)); } else app.kill('SIGTERM'); }
    if (portal) await new Promise(resolve => portal.close(resolve));
    await mongoose.disconnect(); if (database) await database.stop();
    if (checkout) {
      assert.ok(path.resolve(checkout).startsWith(`${path.resolve(root, '.next')}${path.sep}`));
      assert.ok(path.basename(checkout).startsWith('content-browser-'));
      rmSync(checkout, { recursive: true, force: true, maxRetries: 5, retryDelay: 200 });
    }
  }
}
